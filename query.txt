SELECT Id,
       Name,
       Repository__c,
       Branch__c,
       Coverage__c,
       Quality_Gate__c,
       Total_Files__c,
       Total_Violations__c
FROM dx_Code_Analyzer__c
ORDER BY CreatedDate DESC
LIMIT 5

       PR_Title__c,
